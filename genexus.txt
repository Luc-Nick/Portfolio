&ban = false
	java [!&pathname!]=httpContext.getDefaultPath();
	//&pathname="C:/Program Files (x86)/Apache Software Foundation/Tomcat 8.5/webapps/FilesPracticeGx17JavaEnvironment/static/Saves"
	&pathname = &pathname+&File.Separator+"static"+&File.Separator+"Saves"+&File.Separator+"licitaciones"
	&directorio.Source=&pathname
	if &directorio.Exists()
		for &File in &directorio.GetFiles() 
			&Licitacion.Load(&idLic)
			&Existe = true
			for each Licitacion
				where LicitacionURL = &File.GetName() //si ya existe no lo guardara
				&Existe = false
			endfor
			if &Existe
				//&Licitacion.LicitacionNombre = &File.GetName()
				//&folder = "/GPWEBSITE2024JavaEnvironment/static/Saves/licitaciones/"+&File.GetName()
				&folder = &File.GetName()
				&Licitacion.LicitacionURL = &folder
				&Licitacion.LicitacionFechaCarga = ServerDate()
				&DateApertura = &Licitacion.LicitacionFechaApertura 
				&Licitacion.LicitacionFechaBaja = &DateApertura.AddDays(1) //Se desactiva un dia posterior a la apertura
				&Licitacion.Save()
			endif
		endfor
		&ban = true
	endif

	endif
